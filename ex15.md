#include <stdio.h>
#include <string.h>

int main() {
    char users[10][30], files[10][10][30];
    int count[10] = {0};
    int nu = 0, choice, i, u, found;
    char user[30], file[30];

    while (1) {
        printf("\n1.Create User  2.Create File");
        printf("\n3.List Files   4.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Enter username: ");
            scanf("%29s", user);

            found = 0;

            for (i = 0; i < nu; i++)
                if (strcmp(users[i], user) == 0)
                    found = 1;

            if (found)
                printf("User already exists\n");
            else if (nu < 10) {
                strcpy(users[nu], user);
                nu++;
                printf("User directory created\n");
            } else {
                printf("Maximum users reached\n");
            }
        } else if (choice == 2 || choice == 3) {
            printf("Enter username: ");
            scanf("%29s", user);

            u = -1;

            for (i = 0; i < nu; i++)
                if (strcmp(users[i], user) == 0)
                    u = i;

            if (u == -1) {
                printf("User not found\n");
                continue;
            }

            if (choice == 2) {
                if (count[u] >= 10) {
                    printf("Directory is full\n");
                    continue;
                }

                printf("Enter filename: ");
                scanf("%29s", file);

                found = 0;

                for (i = 0; i < count[u]; i++)
                    if (strcmp(files[u][i], file) == 0)
                        found = 1;

                if (found)
                    printf("File already exists\n");
                else {
                    strcpy(files[u][count[u]++], file);
                    printf("File created\n");
                }
            } else {
                printf("Files of %s:\n", user);

                for (i = 0; i < count[u]; i++)
                    printf("%s\n", files[u][i]);
            }
        } else if (choice == 4) {
            break;
        } else {
            printf("Invalid choice\n");
        }
    }

    return 0;
